cmake_minimum_required(VERSION 3.20)
project(LunaAudioAI VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_definitions(_CRT_SECURE_NO_WARNINGS _USE_MATH_DEFINES)
endif()
if(WIN32)
  add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
endif()
option(LUNA_SANITIZE "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
if(LUNA_SANITIZE AND NOT MSVC)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
set(RN third_party/rnnoise)
add_library(rnnoise STATIC ${RN}/src/denoise.c ${RN}/src/rnn.c
  ${RN}/src/rnn_data.c ${RN}/src/pitch.c ${RN}/src/kiss_fft.c ${RN}/src/celt_lpc.c)
target_include_directories(rnnoise PUBLIC ${RN}/include PRIVATE ${RN}/src)
target_compile_definitions(rnnoise PRIVATE RNNOISE_BUILD)
add_library(miniaudio STATIC src/miniaudio_impl.c)
target_include_directories(miniaudio PUBLIC third_party)
target_compile_definitions(miniaudio PUBLIC MA_NO_ENGINE MA_NO_RESOURCE_MANAGER
  MA_NO_NODE_GRAPH MA_NO_GENERATION MA_NO_MP3 MA_NO_FLAC MA_NO_ENCODING
  MA_NO_NULL)
if(WIN32)
  target_compile_definitions(miniaudio PUBLIC MA_ENABLE_ONLY_SPECIFIC_BACKENDS MA_ENABLE_WASAPI)
  target_link_libraries(miniaudio PUBLIC ole32 uuid)
else()
  find_package(Threads REQUIRED)
  target_link_libraries(miniaudio PUBLIC Threads::Threads ${CMAKE_DL_LIBS} m)
endif()
add_library(luna_core STATIC src/engine.cpp src/processor.cpp src/wav.cpp)
target_include_directories(luna_core PUBLIC include)
target_link_libraries(luna_core PUBLIC rnnoise miniaudio)
add_library(luna_audio STATIC src/audio.cpp)
target_link_libraries(luna_audio PUBLIC luna_core)
foreach(t luna_core luna_audio)
  if(MSVC)
    target_compile_options(${t} PRIVATE /W4 /permissive-)
  else()
    target_compile_options(${t} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endforeach()
add_executable(luna-cli src/cli.cpp)
target_link_libraries(luna-cli PRIVATE luna_audio)
if(WIN32)
  target_link_libraries(luna-cli PRIVATE shell32)
  add_executable(LunaAudioAI WIN32 src/windows_ui.cpp)
  if(MINGW)
    enable_language(RC)
    target_sources(LunaAudioAI PRIVATE src/windows_app.rc)
    target_include_directories(LunaAudioAI PRIVATE src)
  elseif(MSVC)
    target_sources(LunaAudioAI PRIVATE src/windows_app.manifest)
  endif()
  target_link_libraries(LunaAudioAI PRIVATE luna_audio comctl32 gdi32 user32)
  if(MINGW)
    target_link_options(LunaAudioAI PRIVATE -municode)
    foreach(t LunaAudioAI luna-cli)
      target_link_options(${t} PRIVATE -static -static-libgcc -static-libstdc++)
    endforeach()
  endif()
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(luna-tests tests/tests.cpp)
  target_link_libraries(luna-tests PRIVATE luna_audio)
  if(MINGW)
    target_link_options(luna-tests PRIVATE -static -static-libgcc -static-libstdc++)
  endif()
  add_test(NAME core_and_wav COMMAND luna-tests)
  add_test(NAME cli_help COMMAND luna-cli --help)
endif()
